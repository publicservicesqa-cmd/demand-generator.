<script>
  let jobsCount = 0;

  function initApp() {
    const randomRef = 'NEP-' + Math.floor(100000 + Math.random() * 900000);
    document.getElementById('app-ref-display').innerText = randomRef;
    
    // Add default initial row
    addJobRow('General Worker / Cleaner', 10, 0, 1000, 300, 'QAR');
  }

  function addJobRow(cat = '', male = 0, female = 0, salary = '', food = '', curr = 'QAR') {
    jobsCount++;
    const tbody = document.getElementById('jobsTbody');
    const tr = document.createElement('tr');
    tr.id = `row-${jobsCount}`;
    tr.innerHTML = `
      <td><input type="text" class="j-cat" value="${cat}" placeholder="e.g. Mason / Electrician" required></td>
      <td><input type="number" class="j-male" value="${male}" min="0" oninput="calculateTotals()" required></td>
      <td><input type="number" class="j-female" value="${female}" min="0" oninput="calculateTotals()" required></td>
      <td><input type="number" class="j-salary" value="${salary}" placeholder="1000" required></td>
      <td><input type="number" class="j-food" value="${food}" placeholder="300"></td>
      <td><input type="text" class="j-curr" value="${curr}" style="text-align:center;"></td>
      <td class="row-actions"><button type="button" class="btn-del" onclick="removeJobRow('row-${jobsCount}')">&times;</button></td>
    `;
    tbody.appendChild(tr);
    calculateTotals();
  }

  function removeJobRow(rowId) {
    const row = document.getElementById(rowId);
    if(row) {
      row.remove();
      calculateTotals();
    }
  }

  function calculateTotals() {
    let maleSum = 0;
    let femaleSum = 0;
    
    const males = document.querySelectorAll('.j-male');
    const females = document.querySelectorAll('.j-female');

    males.forEach(input => maleSum += parseInt(input.value || 0, 10));
    females.forEach(input => femaleSum += parseInt(input.value || 0, 10));

    document.getElementById('total_male').innerText = maleSum;
    document.getElementById('total_female').innerText = femaleSum;
    document.getElementById('total_workers').innerText = maleSum + femaleSum;
  }

  function getFormData() {
    const jobs = [];
    const rows = document.querySelectorAll('#jobsTbody tr');
    
    rows.forEach(tr => {
      jobs.push({
        category: tr.querySelector('.j-cat').value,
        qty_male: parseInt(tr.querySelector('.j-male').value || 0, 10),
        qty_female: parseInt(tr.querySelector('.j-female').value || 0, 10),
        total_qty: parseInt(tr.querySelector('.j-male').value || 0, 10) + parseInt(tr.querySelector('.j-female').value || 0, 10),
        basic_salary: tr.querySelector('.j-salary').value,
        food_allowance: tr.querySelector('.j-food').value || '0',
        currency: tr.querySelector('.j-curr').value || 'QAR'
      });
    });

    return {
      app_ref: document.getElementById('app-ref-display').innerText,
      company_name: document.getElementById('company_name').value,
      cr_number: document.getElementById('cr_number').value,
      company_address: document.getElementById('company_address').value,
      sponsor_name: document.getElementById('sponsor_name').value,
      contact_phone: document.getElementById('contact_phone').value,
      contact_email: document.getElementById('contact_email').value,
      agency_name: document.getElementById('agency_name').value,
      agency_license: document.getElementById('agency_license').value,
      agency_address: document.getElementById('agency_address').value,
      working_hours: document.getElementById('working_hours').value,
      working_days: document.getElementById('working_days').value,
      contract_period: document.getElementById('contract_period').value,
      accommodation: document.getElementById('accommodation').value,
      transportation: document.getElementById('transportation').value,
      air_ticket: document.getElementById('air_ticket').value,
      total_male: document.getElementById('total_male').innerText,
      total_female: document.getElementById('total_female').innerText,
      total_workers: document.getElementById('total_workers').innerText,
      jobs: jobs,
      date_today: new Date().toISOString().split('T')[0]
    };
  }

  function saveDraft() {
    const data = getFormData();
    let drafts = JSON.parse(localStorage.getItem('nepal_embassy_drafts') || '[]');
    drafts = drafts.filter(d => d.app_ref !== data.app_ref);
    drafts.push(data);
    localStorage.setItem('nepal_embassy_drafts', JSON.stringify(drafts));
    showStatus('Draft saved successfully to local storage!', 'ok');
  }

  function openLoadModal() {
    const modal = document.getElementById('loadModal');
    const container = document.getElementById('savedList');
    const drafts = JSON.parse(localStorage.getItem('nepal_embassy_drafts') || '[]');

    if(drafts.length === 0) {
      container.innerHTML = '<p style="color:var(--muted)">No saved drafts available.</p>';
    } else {
      container.innerHTML = drafts.map((d, index) => `
        <div class="app-row">
          <div>
            <b>${d.company_name || 'Unnamed Company'}</b> (${d.app_ref})<br>
            <small style="color:var(--muted)">${d.date_today} | Workers: ${d.total_workers}</small>
          </div>
          <button class="btn btn-secondary" onclick="loadDraft(${index})">Load</button>
        </div>
      `).join('');
    }
    modal.classList.add('open');
  }

  function closeLoadModal() {
    document.getElementById('loadModal').classList.remove('open');
  }

  function loadDraft(index) {
    const drafts = JSON.parse(localStorage.getItem('nepal_embassy_drafts') || '[]');
    const d = drafts[index];
    if(!d) return;

    document.getElementById('app-ref-display').innerText = d.app_ref;
    document.getElementById('company_name').value = d.company_name || '';
    document.getElementById('cr_number').value = d.cr_number || '';
    document.getElementById('company_address').value = d.company_address || '';
    document.getElementById('sponsor_name').value = d.sponsor_name || '';
    document.getElementById('contact_phone').value = d.contact_phone || '';
    document.getElementById('contact_email').value = d.contact_email || '';
    document.getElementById('agency_name').value = d.agency_name || '';
    document.getElementById('agency_license').value = d.agency_license || '';
    document.getElementById('agency_address').value = d.agency_address || '';
    document.getElementById('working_hours').value = d.working_hours || 8;
    document.getElementById('working_days').value = d.working_days || 6;
    document.getElementById('contract_period').value = d.contract_period || 2;
    document.getElementById('accommodation').value = d.accommodation || 'Provided by Company';
    document.getElementById('transportation').value = d.transportation || 'Provided by Company';
    document.getElementById('air_ticket').value = d.air_ticket || 'Joining and Return Provided free by Employer';

    const tbody = document.getElementById('jobsTbody');
    tbody.innerHTML = '';
    jobsCount = 0;

    if(d.jobs && d.jobs.length > 0) {
      d.jobs.forEach(j => {
        addJobRow(j.category, j.qty_male, j.qty_female, j.basic_salary, j.food_allowance, j.currency);
      });
    } else {
      addJobRow();
    }

    closeLoadModal();
    showStatus('Draft loaded successfully!', 'ok');
  }

  function resetForm() {
    if(confirm('Start a new form? Unsaved changes will be cleared.')) {
      document.getElementById('demandForm').reset();
      document.getElementById('jobsTbody').innerHTML = '';
      jobsCount = 0;
      initApp();
      showStatus('Form reset.', 'ok');
    }
  }

  function showStatus(msg, type) {
    const st = document.getElementById('status');
    st.innerText = msg;
    st.className = type;
    st.style.display = 'block';
    setTimeout(() => { st.style.display = 'none'; }, 4000);
  }

  function generateDocuments() {
    const fileInput = document.getElementById('templateFile');
    if (!fileInput.files || fileInput.files.length === 0) {
      alert('Please select a template .docx file to populate.');
      return;
    }

    const data = getFormData();
    const file = fileInput.files[0];
    const reader = new FileReader();

    reader.onload = function(evt) {
      try {
        const content = evt.target.result;
        const zip = new PizZip(content);
        const doc = new window.docxtemplater(zip, {
          paragraphLoop: true,
          linebreaks: true,
        });

        // Set the template data
        doc.render(data);

        const out = doc.getZip().generate({
          type: 'blob',
          mimeType: 'application/vnd.openxmlformats-officedocument.wordprocessingml.document',
        });

        const filename = `Demand_Document_${data.company_name.replace(/[^a-z0-9]/gi, '_')}_${data.app_ref}.docx`;
        saveAs(out, filename);
        showStatus('Document generated and download started successfully!', 'ok');
      } catch (error) {
        console.error('Docxtemplater error:', error);
        showStatus('Error generating document: ' + error.message, 'err');
      }
    };

    reader.readAsArrayBuffer(file);
  }

  window.onload = initApp;
</script>
